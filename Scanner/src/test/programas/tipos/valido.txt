bool ehPrimo(int x) {
  if (x % 2 == 0) {
    return false;
  }
  for (int i = 3; i < x; i += 2;) {
    if (x % i == 0) {
      return false;
    }
  }
  return true;
}

int mdc(int a, int b) {
  if (b == 0) {
    return a;
  } else {
    return mdc(b, a % b);
  }
}

float media(int a, float b) {
  return (a + b) / 2;
}
