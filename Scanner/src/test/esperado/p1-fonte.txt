@descobre se um numero eh primo
bool ehPrimo(int x) {
  if (x % 2 == 0) {
    return false;
  }
  for (int i = 3; i < x; i += 2;) {
    if (x % i == 0) {
      return false;
    }
  }
  return true;
}

@imprimir primos ate 1000
void main() {
  int i = 1;
  while (i <= 1000) {
    if (ehPrimo(i)) {
      print(i);
      i += 1;
    }
  }
}
